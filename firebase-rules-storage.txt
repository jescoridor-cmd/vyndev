rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    match /plugins/{fileName} {
      // Membaca file hanya untuk pengguna yang sudah login.
      allow read: if request.auth != null;

      // Upload hanya untuk akun dengan dokumen members dan role admin.
      allow write: if request.auth != null
        && firestore.exists(/databases/(default)/documents/members/$(request.auth.uid))
        && firestore.get(/databases/(default)/documents/members/$(request.auth.uid)).data.role == 'admin'
        && request.resource.size < 100 * 1024 * 1024
        && request.resource.contentType == 'application/java-archive';
    }
  }
}
